### 문제의 복잡성
- 26개 케이스별 매핑 규칙 필요
- 지원본부 특별 처리 로직
- employeeId 기반 예외 케이스 다수
- LVL 기반 + 직책 기반 혼합 매핑

## 해결 과정

### 1단계: 지원본부 특별 처리 로직 (3월 중순)
```javascript
const isSupportHeadquarters = 
  item.headquarters?.replace(/\s+/g, '') === '지원본부'

const mapLevelToInterviewRank = (level?: string | null): string | null => {
  const levelNo = (level || '').replace(/^LVL/i, '')
  const levelMap: Record<string, string> = {
    '1': '담당', '2': '소장', '3': '지사장', '4': '본부장',
    '6': '사장', '7': '회장', '8': '실장', '9': '부사장'
  }
  return levelMap[levelNo] || null
}
```

### 2단계: 특정 사원번호 예외 처리 (3월 말)
```javascript
// 특정 사원번호는 무조건 실장으로 처리
if (processed.employeeId === '0600198') {
  processed.dutyNm = '실장'
  processed.positionNm = '실장'
  processed.positionCode = '실장'
}
```

### 3단계: 직책별 세분화 매핑 (4월 초)
```javascript
if (processed.dutyCd === '105' || duty === '본부장') {
  const rankName = isSupportHeadquarters ? '실장' : '본부장'
  processed.dutyNm = rankName
  processed.positionNm = rankName
  processed.positionCode = rankName
} else if (duty === '영업소장') {
  // 영업소장 -> 소장(과장) 컬럼으로 매핑
  processed.dutyNm = '소장'
}
```
